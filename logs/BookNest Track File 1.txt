Database Decision: Supabase over Firebase
For a messaging app, Supabase (PostgreSQL) is the better choice:
Relational integrity for chats, messages, users, memberships
Row Level Security (RLS) for fine-grained access control
Real-time subscriptions built-in (LISTEN/NOTIFY via WebSockets)
No vendor lock-in — it's just Postgres
Better for complex queries (message search, pagination, analytics)
Firebase's document model becomes painful for relational data like chat memberships and message threading.
Core Schema (Supabase PostgreSQL)
sql
-- Users (extends Supabase auth.users)
create table profiles (
  id uuid references auth.users on delete cascade primary key,
  username text unique not null,
  display_name text,
  avatar_url text,
  status text default 'offline',
  last_seen timestamptz default now(),
  created_at timestamptz default now()
);

-- Conversations (1-on-1 or group)
create table conversations (
  id uuid default gen_random_uuid() primary key,
  type text check (type in ('direct', 'group', 'channel')),
  title text,
  avatar_url text,
  created_by uuid references profiles(id),
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- Conversation members
create table conversation_members (
  id uuid default gen_random_uuid() primary key,
  conversation_id uuid references conversations(id) on delete cascade,
  user_id uuid references profiles(id) on delete cascade,
  role text default 'member' check (role in ('admin', 'moderator', 'member')),
  joined_at timestamptz default now(),
  last_read_message_id uuid,
  unique(conversation_id, user_id)
);

-- Messages
create table messages (
  id uuid default gen_random_uuid() primary key,
  conversation_id uuid references conversations(id) on delete cascade,
  sender_id uuid references profiles(id),
  type text default 'text' check (type in ('text', 'image', 'file', 'system')),
  content text,
  metadata jsonb, -- file URLs, dimensions, etc.
  reply_to uuid references messages(id),
  edited_at timestamptz,
  created_at timestamptz default now()
);

-- Enable RLS
alter table profiles enable row level security;
alter table conversations enable row level security;
alter table conversation_members enable row level security;
alter table messages enable row level security;

-- RLS: Users can only see messages in conversations they're members of
create policy "Users can view messages in their conversations"
  on messages for select
  using (
    exists (
      select 1 from conversation_members
      where conversation_id = messages.conversation_id
      and user_id = auth.uid()
    )

Phase 2: Real-Time Messaging Architecture
Option A: Socket.IO (Recommended for starting out)
Pros: Easy to use, fallbacks, room management
Cons: Single server limitation (use Redis adapter for scaling)
JavaScript
// server/socket/index.js
const { Server } = require('socket.io');
const { createAdapter } = require('@socket.io/redis-adapter');

const initSocket = (httpServer) => {
  const io = new Server(httpServer, {
    cors: { origin: process.env.CLIENT_URL },
    transports: ['websocket', 'polling']
  });

  // Redis adapter for multi-server scaling
  const pubClient = new Redis(process.env.REDIS_URL);
  const subClient = pubClient.duplicate();
  io.adapter(createAdapter(pubClient, subClient));

  // Authentication middleware
  io.use(async (socket, next) => {
    const token = socket.handshake.auth.token;
    try {
      const { data: { user } } = await supabase.auth.getUser(token);
      socket.userId = user.id;
      next();
    } catch (err) {
      next(new Error('Authentication error'));
    }
  });

  io.on('connection', (socket) => {
    // Join personal room for direct messages
    socket.join(`user:${socket.userId}`);
    
    // Join conversation rooms
    socket.on('join_conversation', (conversationId) => {
      socket.join(`conv:${conversationId}`);
    });

    socket.on('leave_conversation', (conversationId) => {
      socket.leave(`conv:${conversationId}`);
    });

    // Handle messages
    socket.on('send_message', async (data, callback) => {
      const { conversationId, content, type = 'text' } = data;
      
      // Verify membership via RLS/DB check
      const { data: member } = await supabase
        .from('conversation_members')
        .select('id')
        .eq('conversation_id', conversationId)
        .eq('user_id', socket.userId)
        .single();

      if (!member) return callback({ error: 'Not a member' });

      // Save to DB
      const { data: message, error } = await supabase
        .from('messages')
        .insert({
          conversation_id: conversationId,
          sender_id: socket.userId,
          content,
          type
        })
        .select('*, sender:profiles(id, username, avatar_url)')
        .single();

      if (error) return callback({ error: error.message });

      // Broadcast to conversation room
      io.to(`conv:${conversationId}`).emit('new_message', message);
      
      // Update conversation timestamp
      await supabase
        .from('conversations')
        .update({ updated_at: new Date() })
        .eq('id', conversationId);

      callback({ success: true, message });
    });

    // Typing indicators
    socket.on('typing', ({ conversationId, isTyping }) => {
      socket.to(`conv:${conversationId}`).emit('user_typing', {
        userId: socket.userId,
        isTyping
      });
    });

    // Presence updates
    socket.on('disconnect', async () => {
      await supabase
        .from('profiles')
        .update({ status: 'offline', last_seen: new Date() })
        .eq('id', socket.userId);
      
      io.emit('user_status', { userId: socket.userId, status: 'offline' });
    });
  });

  return io;
};

import { createClient } from '@supabase/supabase-js';

const supabase = createClient(

// Sign up with email + username
export const signUp = async (email, password, username) => {
  const { data, error } = await supabase.auth.signUp({
    email,
    password,
    options: {
      data: { username }
    }
  });
  
  if (data.user) {
    // Create profile record
    await supabase.from('profiles').insert({
      id: data.user.id,
      username,
      display_name: username
    });
  }
  
  return { data, error };
};

// server/routes/payments.js
const paystack = require('paystack')(process.env.PAYSTACK_SECRET_KEY);

router.post('/initialize', authenticate, async (req, res) => {
  const { amount, email } = req.body;
  
  try {
    const response = await paystack.transaction.initialize({
      amount: amount * 100, // Paystack uses kobo/cents
      email,
      metadata: { userId: req.user.id },
      callback_url: `${process.env.CLIENT_URL}/payment/verify`
    });
    
    res.json(response.data);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// Webhook for payment confirmation
router.post('/webhook', async (req, res) => {
  const hash = crypto
    .createHmac('sha512', process.env.PAYSTACK_SECRET_KEY)
    .update(JSON.stringify(req.body))
    .digest('hex');
    
  if (hash !== req.headers['x-paystack-signature']) {
    return res.status(400).send('Invalid signature');
  }
  
  const event = req.body;
  if (event.event === 'charge.success') {
    // Update user's wallet/credits
    await supabase.from('transactions').insert({
      user_id: event.data.metadata.userId,
      amount: event.data.amount / 100,
      reference: event.data.reference,
      status: 'completed'
    });
  }
  
  res.sendStatus(200);
});

Step 1: Supabase Database Setup
Go to your Supabase dashboard → SQL Editor → New Query. Run this entire block:
sql
-- ============================================
-- TETHER CORE DATABASE SCHEMA
-- ============================================

-- Enable required extensions
create extension if not exists "uuid-ossp";

-- ============================================
-- PROFILES (extends Supabase auth.users)
-- ============================================
create table profiles (
  id uuid references auth.users on delete cascade primary key,
  username text unique not null,
  display_name text,
  phone_number text,
  avatar_url text,
  bio text default '',
  status text default 'offline' check (status in ('online', 'offline', 'away')),
  last_seen timestamptz default now(),
  created_at timestamptz default now()
);

-- ============================================
-- CONVERSATIONS
-- ============================================
create table conversations (
  id uuid default uuid_generate_v4() primary key,
  type text default 'direct' check (type in ('direct', 'group')),
  title text,
  avatar_url text,
  created_by uuid references profiles(id),
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

-- ============================================
-- CONVERSATION MEMBERS
-- ============================================
create table conversation_members (
  id uuid default uuid_generate_v4() primary key,
  conversation_id uuid references conversations(id) on delete cascade,
  user_id uuid references profiles(id) on delete cascade,
  role text default 'member' check (role in ('admin', 'member')),
  joined_at timestamptz default now(),
  last_read_at timestamptz,
  unique(conversation_id, user_id)
);

-- ============================================
-- MESSAGES
-- ============================================
create table messages (
  id uuid default uuid_generate_v4() primary key,
  conversation_id uuid references conversations(id) on delete cascade,
  sender_id uuid references profiles(id),
  content text not null,
  type text default 'text' check (type in ('text', 'image', 'file')),
  metadata jsonb default '{}',
  reply_to uuid references messages(id),
  edited_at timestamptz,
  created_at timestamptz default now()
);

-- ============================================
-- CONTACTS / FOLLOWERS
-- ============================================
create table contacts (
  id uuid default uuid_generate_v4() primary key,
  user_id uuid references profiles(id) on delete cascade,
  contact_id uuid references profiles(id) on delete cascade,
  status text default 'pending' check (status in ('pending', 'accepted', 'blocked')),
  created_at timestamptz default now(),
  unique(user_id, contact_id)
);

-- ============================================
-- TYPING INDICATORS (temporary, auto-cleanup)
-- ============================================
create table typing_indicators (
  id uuid default uuid_generate_v4() primary key,
  conversation_id uuid references conversations(id) on delete cascade,
  user_id uuid references profiles(id) on delete cascade,
  started_at timestamptz default now(),
  unique(conversation_id, user_id)
);

-- ============================================
-- ROW LEVEL SECURITY (RLS)
-- ============================================

-- Profiles: users can read all, update only their own
alter table profiles enable row level security;

create policy "Profiles are viewable by everyone"
  on profiles for select
  using (true);

create policy "Users can update own profile"
  on profiles for update
  using (auth.uid() = id);

-- Conversations: members can see their conversations
alter table conversations enable row level security;

create policy "Members can view conversations"
  on conversations for select
  using (
    exists (
      select 1 from conversation_members
      where conversation_id = conversations.id
      and user_id = auth.uid()
    )
  );

-- Messages: members can see messages in their conversations
alter table messages enable row level security;

create policy "Members can view messages"
  on messages for select
  using (
    exists (
      select 1 from conversation_members
      where conversation_id = messages.conversation_id
      and user_id = auth.uid()
    )
  );

create policy "Members can insert messages"
  on messages for insert
  with check (
    exists (
      select 1 from conversation_members
      where conversation_id = messages.conversation_id
      and user_id = auth.uid()
    )
  );

-- Conversation members: viewable by members
alter table conversation_members enable row level security;

create policy "Members can view membership"
  on conversation_members for select
  using (
    conversation_id in (
      select conversation_id from conversation_members
      where user_id = auth.uid()
    )
  );

-- Contacts: users can see their own contacts
alter table contacts enable row level security;

create policy "Users can view own contacts"
  on contacts for select
  using (user_id = auth.uid() or contact_id = auth.uid());

-- Typing indicators: viewable by conversation members
alter table typing_indicators enable row level security;

create policy "Members can view typing"
  on typing_indicators for select
  using (
    exists (
      select 1 from conversation_members
      where conversation_id = typing_indicators.conversation_id
      and user_id = auth.uid()
    )
  );

-- ============================================
-- REALTIME ENABLEMENT
-- ============================================
alter publication supabase_realtime add table messages;
alter publication supabase_realtime add table typing_indicators;
alter publication supabase_realtime add table conversation_members;

-- ============================================
-- FUNCTIONS & TRIGGERS
-- ============================================

-- Auto-update conversation timestamp on new message
create or replace function update_conversation_timestamp()
returns trigger as $$
begin
  update conversations
  set updated_at = now()
  where id = new.conversation_id;
  return new;
end;
$$ language plpgsql security definer;

create trigger on_message_sent
  after insert on messages
  for each row
  execute function update_conversation_timestamp();

-- Auto-create profile on signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.profiles (id, username, display_name, phone_number)
  values (
    new.id,
    new.raw_user_meta_data->>'username',
    new.raw_user_meta_data->>'display_name',
    new.raw_user_meta_data->>'phone_number'
  );
  return new;
end;
$$ language plpgsql security definer;

create trigger on_auth_user_created
  after insert on auth.users
  for each row
  execute function public.handle_new_user();

-- Cleanup old typing indicators (every 5 minutes)
create or replace function cleanup_typing_indicators()
returns void as $$
begin
  delete from typing_indicators
  where started_at < now() - interval '30 seconds';
end;
$$ language plpgsql;

5. Gem Spending Logic (Backend Function)
When a user reads past 3/7 of a book, they spend gems. Here's the logic:
sql
-- Function to check if user can read more words
create or replace function can_read_words(
  p_user_id uuid,
  p_book_id uuid,
  p_words_to_read int
) returns boolean as $$
declare
  v_total_words int;
  v_current_word int;
  v_free_limit int;
  v_words_needed int;
  v_user_gems int;
  v_gems_required int;
begin
  -- Get book info
  select total_words into v_total_words from club_books where id = p_book_id;
  
  -- Get user progress
  select current_word into v_current_word 
  from user_book_progress 
  where user_id = p_user_id and club_book_id = p_book_id;
  
  if v_current_word is null then
    v_current_word := 0;
  end if;
  
  -- Free limit = 3/7 of book
  v_free_limit := (v_total_words * 3) / 7;
  
  -- If still within free limit, allow
  if (v_current_word + p_words_to_read) <= v_free_limit then
    return true;
  end if;
  
  -- Calculate paid words needed
  v_words_needed := greatest(0, (v_current_word + p_words_to_read) - v_free_limit);
  
  -- 100 words = 1 gem
  v_gems_required := ceil(v_words_needed::float / 100);
  
  -- Check user gems
  select gems into v_user_gems from profiles where id = p_user_id;
  
  return v_user_gems >= v_gems_required;
end;
$$ language plpgsql security definer;

-- Function to spend gems and unlock words
create or replace function spend_gems_for_reading(
  p_user_id uuid,
  p_book_id uuid,
  p_words_to_read int
) returns jsonb as $$
declare
  v_total_words int;
  v_current_word int;
  v_free_limit int;
  v_words_needed int;
  v_gems_required int;
  v_user_gems int;
  v_new_word int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select current_word into v_current_word from user_book_progress where user_id = p_user_id and club_book_id = p_book_id;
  
  if v_current_word is null then v_current_word := 0; end if;
  
  v_free_limit := (v_total_words * 3) / 7;
  v_new_word := least(v_current_word + p_words_to_read, v_total_words);
  
  if v_new_word <= v_free_limit then
    -- Still free, just update progress
    insert into user_book_progress (user_id, club_book_id, current_word, percent_complete)
    values (p_user_id, p_book_id, v_new_word, (v_new_word * 100) / v_total_words)
    on conflict (user_id, club_book_id) 
    do update set current_word = v_new_word, percent_complete = (v_new_word * 100) / v_total_words;
    
    return jsonb_build_object('success', true, 'gems_spent', 0, 'free', true);
  end if;
  
  -- Need gems
  v_words_needed := v_new_word - greatest(v_current_word, v_free_limit);
  v_gems_required := ceil(v_words_needed::float / 100);
  
  select gems into v_user_gems from profiles where id = p_user_id;
  
  if v_user_gems < v_gems_required then
    return jsonb_build_object('success', false, 'error', 'Insufficient gems', 'gems_required', v_gems_required, 'gems_available', v_user_gems);
  end if;
  
  -- Deduct gems
  update profiles set gems = gems - v_gems_required, total_gems_spent = total_gems_spent + v_gems_required where id = p_user_id;
  
  -- Record transaction
  insert into gem_transactions (user_id, type, amount, description)
  values (p_user_id, 'read_unlock', -v_gems_required, 'Unlocked ' || v_words_needed || ' words in book');
  
  -- Update progress
  insert into user_book_progress (user_id, club_book_id, current_word, percent_complete, gems_spent, is_locked)
  values (p_user_id, p_book_id, v_new_word, (v_new_word * 100) / v_total_words, v_gems_required, false)
  on conflict (user_id, club_book_id) 
  do update set 
    current_word = v_new_word, 
    percent_complete = (v_new_word * 100) / v_total_words,
    gems_spent = user_book_progress.gems_spent + v_gems_required,
    is_locked = false;
  
  return jsonb_build_object('success', true, 'gems_spent', v_gems_required, 'free', false, 'new_position', v_new_word);
end;
$$ language plpgsql security definer;

-- Function to check if user can read more words
create or replace function can_read_words(
  p_user_id uuid,
  p_book_id uuid,
  p_words_to_read int
) returns boolean as $$
declare
  v_total_words int;
  v_current_word int;
  v_free_limit int;
  v_words_needed int;
  v_user_gems int;
  v_gems_required int;
begin
  -- Get book info
  select total_words into v_total_words from club_books where id = p_book_id;
  
  -- Get user progress
  select current_word into v_current_word 
  from user_book_progress 
  where user_id = p_user_id and club_book_id = p_book_id;
  
  if v_current_word is null then
    v_current_word := 0;
  end if;
  
  -- Free limit = 3/7 of book
  v_free_limit := (v_total_words * 3) / 7;
  
  -- If still within free limit, allow
  if (v_current_word + p_words_to_read) <= v_free_limit then
    return true;
  end if;
  
  -- Calculate paid words needed
  v_words_needed := greatest(0, (v_current_word + p_words_to_read) - v_free_limit);
  
  -- 100 words = 1 gem
  v_gems_required := ceil(v_words_needed::float / 100);
  
  -- Check user gems
  select gems into v_user_gems from profiles where id = p_user_id;
  
  return v_user_gems >= v_gems_required;
end;
$$ language plpgsql security definer;

-- Function to spend gems and unlock words
create or replace function spend_gems_for_reading(
  p_user_id uuid,
  p_book_id uuid,
  p_words_to_read int
) returns jsonb as $$
declare
  v_total_words int;
  v_current_word int;
  v_free_limit int;
  v_words_needed int;
  v_gems_required int;
  v_user_gems int;
  v_new_word int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select current_word into v_current_word from user_book_progress where user_id = p_user_id and club_book_id = p_book_id;
  
  if v_current_word is null then v_current_word := 0; end if;
  
  v_free_limit := (v_total_words * 3) / 7;
  v_new_word := least(v_current_word + p_words_to_read, v_total_words);
  
  if v_new_word <= v_free_limit then
    -- Still free, just update progress
    insert into user_book_progress (user_id, club_book_id, current_word, percent_complete)
    values (p_user_id, p_book_id, v_new_word, (v_new_word * 100) / v_total_words)
    on conflict (user_id, club_book_id) 
    do update set current_word = v_new_word, percent_complete = (v_new_word * 100) / v_total_words;
    
    return jsonb_build_object('success', true, 'gems_spent', 0, 'free', true);
  end if;
  
  -- Need gems
  v_words_needed := v_new_word - greatest(v_current_word, v_free_limit);
  v_gems_required := ceil(v_words_needed::float / 100);
  
  select gems into v_user_gems from profiles where id = p_user_id;
  
  if v_user_gems < v_gems_required then
    return jsonb_build_object('success', false, 'error', 'Insufficient gems', 'gems_required', v_gems_required, 'gems_available', v_user_gems);
  end if;
  
  -- Deduct gems
  update profiles set gems = gems - v_gems_required, total_gems_spent = total_gems_spent + v_gems_required where id = p_user_id;
  
  -- Record transaction
  insert into gem_transactions (user_id, type, amount, description)
  values (p_user_id, 'read_unlock', -v_gems_required, 'Unlocked ' || v_words_needed || ' words in book');
  
  -- Update progress
  insert into user_book_progress (user_id, club_book_id, current_word, percent_complete, gems_spent, is_locked)
  values (p_user_id, p_book_id, v_new_word, (v_new_word * 100) / v_total_words, v_gems_required, false)
  on conflict (user_id, club_book_id) 
  do update set 
    current_word = v_new_word, 
    percent_complete = (v_new_word * 100) / v_total_words,
    gems_spent = user_book_progress.gems_spent + v_gems_required,
    is_locked = false;
  
  return jsonb_build_object('success', true, 'gems_spent', v_gems_required, 'free', false, 'new_position', v_new_word);
end;
$$ language plpgsql security definer;

  after insert or update or delete on book_chapters

-- Get words before chapter
create or replace function get_words_before_chapter(
  p_book_id uuid,
  p_chapter_number int
) returns int as $$
declare
  v_total int;
begin
  select coalesce(sum(word_count), 0) into v_total
  from book_chapters
  where club_book_id = p_book_id and chapter_number < p_chapter_number;
  return v_total;
end;
$$ language plpgsql security definer;

-- Can read chapter words
create or replace function can_read_chapter_words(
  p_user_id uuid,
  p_book_id uuid,
  p_chapter_number int,
  p_words_in_chapter int
) returns boolean as $$
declare
  v_total_words int;
  v_words_before int;
  v_global_word int;
  v_free_limit int;
  v_words_needed int;
  v_user_gems int;
  v_gems_required int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select get_words_before_chapter(p_book_id, p_chapter_number) into v_words_before;
  v_global_word := v_words_before + p_words_in_chapter;
  v_free_limit := (v_total_words * 3) / 7;
  
  if v_global_word <= v_free_limit then
    return true;
  end if;
  
  v_words_needed := v_global_word - v_free_limit;
  v_gems_required := ceil(v_words_needed::float / 100);
  select gems into v_user_gems from profiles where id = p_user_id;
  
  return v_user_gems >= v_gems_required;
end;
$$ language plpgsql security definer;

-- Spend gems for chapter reading
create or replace function spend_gems_for_chapter_reading(
  p_user_id uuid,
  p_book_id uuid,
  p_chapter_number int,
  p_words_in_chapter int
) returns jsonb as $$
declare
  v_total_words int;
  v_words_before int;
  v_global_word int;
  v_free_limit int;
  v_words_needed int;
  v_gems_required int;
  v_user_gems int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select get_words_before_chapter(p_book_id, p_chapter_number) into v_words_before;
  
  v_global_word := v_words_before + p_words_in_chapter;
  v_free_limit := (v_total_words * 3) / 7;
  
  if v_global_word <= v_free_limit then
    insert into user_book_progress (user_id, club_book_id, current_chapter, current_word_in_chapter, current_word, percent_complete)
    values (p_user_id, p_book_id, p_chapter_number, p_words_in_chapter, v_global_word, (v_global_word * 100) / v_total_words)
    on conflict (user_id, club_book_id) 
    do update set 
      current_chapter = p_chapter_number,
      current_word_in_chapter = p_words_in_chapter,
      current_word = v_global_word,
      percent_complete = (v_global_word * 100) / v_total_words;
    
    return jsonb_build_object('success', true, 'gems_spent', 0, 'free', true, 'global_word', v_global_word);
  end if;
  
  v_words_needed := v_global_word - v_free_limit;
  v_gems_required := ceil(v_words_needed::float / 100);
  select gems into v_user_gems from profiles where id = p_user_id;
  
  if v_user_gems < v_gems_required then
    return jsonb_build_object('success', false, 'error', 'Insufficient gems', 'gems_required', v_gems_required, 'gems_available', v_user_gems);
  end if;
  
  update profiles set gems = gems - v_gems_required, total_gems_spent = total_gems_spent + v_gems_required where id = p_user_id;
  
  insert into gem_transactions (user_id, type, amount, description)
  values (p_user_id, 'read_unlock', -v_gems_required, 'Unlocked chapter ' || p_chapter_number);
  
  insert into user_book_progress (user_id, club_book_id, current_chapter, current_word_in_chapter, current_word, percent_complete, gems_spent, is_locked)
  values (p_user_id, p_book_id, p_chapter_number, p_words_in_chapter, v_global_word, (v_global_word * 100) / v_total_words, v_gems_required, false)
  on conflict (user_id, club_book_id) 
  do update set 
    current_chapter = p_chapter_number,
    current_word_in_chapter = p_words_in_chapter,
    current_word = v_global_word,
    percent_complete = (v_global_word * 100) / v_total_words,
    gems_spent = user_book_progress.gems_spent + v_gems_required,
    is_locked = false;
  
  return jsonb_build_object('success', true, 'gems_spent', v_gems_required, 'free', false, 'global_word', v_global_word);
end;
$$ language plpgsql security definer;

returns trigger as $$
begin
  update club_books
  set total_words = (select coalesce(sum(word_count), 0) from book_chapters where club_book_id = coalesce(new.club_book_id, old.club_book_id))

  return coalesce(new, old);
end;

-- Get words before chapter
create or replace function get_words_before_chapter(
  p_book_id uuid,
  p_chapter_number int
) returns int as $$
declare
  v_total int;
begin
  select coalesce(sum(word_count), 0) into v_total
  from book_chapters
  where club_book_id = p_book_id and chapter_number < p_chapter_number;
  return v_total;
end;
$$ language plpgsql security definer;

-- Can read chapter words
create or replace function can_read_chapter_words(
  p_user_id uuid,
  p_book_id uuid,
  p_chapter_number int,
  p_words_in_chapter int
) returns boolean as $$
declare
  v_total_words int;
  v_words_before int;
  v_global_word int;
  v_free_limit int;
  v_words_needed int;
  v_user_gems int;
  v_gems_required int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select get_words_before_chapter(p_book_id, p_chapter_number) into v_words_before;
  
  v_global_word := v_words_before + p_words_in_chapter;
  v_free_limit := (v_total_words * 3) / 7;
  
  if v_global_word <= v_free_limit then
    return true;
  end if;
  
  v_words_needed := v_global_word - v_free_limit;
  v_gems_required := ceil(v_words_needed::float / 100);
  
  select gems into v_user_gems from profiles where id = p_user_id;
  
  return v_user_gems >= v_gems_required;
end;
$$ language plpgsql security definer;

-- Spend gems for chapter reading
create or replace function spend_gems_for_chapter_reading(
  p_user_id uuid,
  p_book_id uuid,
  p_chapter_number int,
  p_words_in_chapter int
) returns jsonb as $$
declare
  v_total_words int;
  v_words_before int;
  v_global_word int;
  v_free_limit int;
  v_words_needed int;
  v_gems_required int;
  v_user_gems int;
begin
  select total_words into v_total_words from club_books where id = p_book_id;
  select get_words_before_chapter(p_book_id, p_chapter_number) into v_words_before;
  
  v_global_word := v_words_before + p_words_in_chapter;
  v_free_limit := (v_total_words * 3) / 7;
  
  if v_global_word <= v_free_limit then
    insert into user_book_progress (user_id, club_book_id, current_chapter, current_word_in_chapter, current_word, percent_complete)
    values (p_user_id, p_book_id, p_chapter_number, p_words_in_chapter, v_global_word, (v_global_word * 100) / v_total_words)
    on conflict (user_id, club_book_id) 
    do update set 
      current_chapter = p_chapter_number,
      current_word_in_chapter = p_words_in_chapter,
      current_word = v_global_word,
      percent_complete = (v_global_word * 100) / v_total_words;
    
    return jsonb_build_object('success', true, 'gems_spent', 0, 'free', true, 'global_word', v_global_word);
  end if;
  
  v_words_needed := v_global_word - v_free_limit;
  v_gems_required := ceil(v_words_needed::float / 100);
  
  select gems into v_user_gems from profiles where id = p_user_id;
  
  if v_user_gems < v_gems_required then
    return jsonb_build_object('success', false, 'error', 'Insufficient gems', 'gems_required', v_gems_required, 'gems_available', v_user_gems);
  end if;
  
  update profiles set gems = gems - v_gems_required, total_gems_spent = total_gems_spent + v_gems_required where id = p_user_id;
  
  insert into gem_transactions (user_id, type, amount, description)
  values (p_user_id, 'read_unlock', -v_gems_required, 'Unlocked chapter ' || p_chapter_number);
  
  insert into user_book_progress (user_id, club_book_id, current_chapter, current_word_in_chapter, current_word, percent_complete, gems_spent, is_locked)
  values (p_user_id, p_book_id, p_chapter_number, p_words_in_chapter, v_global_word, (v_global_word * 100) / v_total_words, v_gems_required, false)
  on conflict (user_id, club_book_id) 
  do update set 
    current_chapter = p_chapter_number,
    current_word_in_chapter = p_words_in_chapter,
    current_word = v_global_word,
    percent_complete = (v_global_word * 100) / v_total_words,
    gems_spent = user_book_progress.gems_spent + v_gems_required,
    is_locked = false;
  
  return jsonb_build_object('success', true, 'gems_spent', v_gems_required, 'free', false, 'global_word', v_global_word);
end;
$$ language plpgsql security definer;

private val LightColorScheme = lightColorScheme(

    tertiary = LightTextPrimary,
    background = LightBackground,
    surface = LightChatBackground,
    surfaceVariant = LightReceivedMessage,

    outline = LightBorder

@Composable
fun TetherTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    dynamicColor: Boolean = false,
    content: @Composable () -> Unit
) {
    val colorScheme = when {
        dynamicColor && Build.VERSION.SDK_INT >= Build.VERSION_CODES.S -> {
            val context = LocalContext.current
            if (darkTheme) dynamicDarkColorScheme(context) else dynamicLightColorScheme(context)
        }
        darkTheme -> DarkColorScheme
        else -> LightColorScheme
    }
    
    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            window.statusBarColor = colorScheme.background.toArgb()
            WindowCompat.getInsetsController(window, view).isAppearanceLightStatusBars = !darkTheme
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = TetherTypography,
        content = content
    )
}

  late Animation<double> _scaleAnimation;

  @override
  void initState() {
    super.initState();
    
    _controller = AnimationController(
      duration: const Duration(milliseconds: 2000),
      vsync: this,
    );

    _fadeAnimation = Tween<double>(begin: 0.0, end: 1.0).animate(
      CurvedAnimation(
        parent: _controller,
        curve: const Interval(0.0, 0.5, curve: Curves.easeIn),
      ),
    );

    _scaleAnimation = Tween<double>(begin: 0.8, end: 1.0).animate(
      CurvedAnimation(
        parent: _controller,
        curve: const Interval(0.0, 0.5, curve: Curves.easeOut),
      ),
    );

    _controller.forward();

    // Check auth status after animation
    Future.delayed(const Duration(seconds: 3), () {
      _checkAuth();
    });
  }

  Future<void> _checkAuth() async {
    final supabase = SupabaseService();
    final session = supabase.auth.currentSession;

    if (!mounted) return;

    if (session != null) {
      // User is logged in → go to clubs
      context.go('/clubs');
    } else {
      // User is not logged in → go to login
      context.go('/login');
    }
  }

  void dispose() {
    _controller.dispose();

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';

class TetherBottomNav extends StatefulWidget {
  final int currentIndex;
  final Function(int) onTap;

  const TetherBottomNav({
    super.key,
    required this.currentIndex,
    required this.onTap,
  });

  @override
  State<TetherBottomNav> createState() => _TetherBottomNavState();
}

class _TetherBottomNavState extends State<TetherBottomNav>
    with SingleTickerProviderStateMixin {
  late AnimationController _centerFabController;
  late Animation<double> _centerFabScale;

  // Tab configurations
  final List<_NavTab> _tabs = const [
    _NavTab(icon: Icons.view_agenda_outlined, label: 'Feed'),
    _NavTab(icon: Icons.explore_outlined, label: 'Discover'),
    _NavTab(icon: Icons.menu_book, label: 'Books', isCenter: true),
    _NavTab(icon: Icons.chat_bubble_outline, label: 'Messages'),
    _NavTab(icon: Icons.person_outline, label: 'Profile'),
  ];

  @override
  void initState() {
    super.initState();
    _centerFabController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 220),
    );
    _centerFabScale = Tween<double>(begin: 1.0, end: 1.12).animate(
      CurvedAnimation(
        parent: _centerFabController,
        curve: Curves.easeOutBack,
      ),
    );
  }

  @override
  void dispose() {
    _centerFabController.dispose();
    super.dispose();
  }

  void _onCenterTap() {
    HapticFeedback.mediumImpact();
    _centerFabController.forward().then((_) {
      _centerFabController.reverse();
    });
    widget.onTap(2);
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      height: 100, // Extra space for center FAB protrusion
      alignment: Alignment.bottomCenter,
      child: Stack(
        alignment: Alignment.bottomCenter,
        clipBehavior: Clip.none,
        children: [
          // Main floating bar container
          Container(
            margin: const EdgeInsets.symmetric(horizontal: 16, vertical: 12),
            height: 64,
            decoration: BoxDecoration(
              color: const Color(0xFF0A0A0A),
              borderRadius: BorderRadius.circular(24),
              boxShadow: [
                BoxShadow(
                  color: Colors.black.withOpacity(0.4),
                  blurRadius: 20,
                  offset: const Offset(0, 8),
                ),
                BoxShadow(
                  color: Colors.black.withOpacity(0.2),
                  blurRadius: 40,
                  offset: const Offset(0, 4),
                ),
              ],
            ),
            child: Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: List.generate(5, (index) {
                if (index == 2) {
                  // Center slot - spacer for FAB cutout
                  return const SizedBox(width: 56);
                }
                return _buildTab(index);
              }),
            ),
          ),

          // Center FAB with inverted scoop effect
          Positioned(
            top: -8, // Protrudes 20dp above the 64dp bar (64 - 56 = 8, then push up)
            child: GestureDetector(
              onTap: _onCenterTap,
              child: AnimatedBuilder(
                animation: _centerFabScale,
                builder: (context, child) {
                  return Transform.scale(
                    scale: widget.currentIndex == 2
                        ? _centerFabScale.value
                        : 1.0,
                    child: Container(
                      width: 56,
                      height: 56,
                      decoration: BoxDecoration(
                        shape: BoxShape.circle,
                        gradient: const LinearGradient(
                          begin: Alignment.topCenter,
                          end: Alignment.bottomCenter,
                          colors: [
                            Color(0xFFFFD000), // Vivid Yellow
                            Color(0xFFFF6A00), // Warm Deep Orange
                          ],
                          stops: [0.0, 1.0],
                        ),
                        boxShadow: [
                          BoxShadow(
                            color: const Color(0xFFFF6A00).withOpacity(0.4),
                            blurRadius: 16,
                            offset: const Offset(0, 6),
                          ),
                          BoxShadow(
                            color: const Color(0xFFFFD000).withOpacity(0.2),
                            blurRadius: 8,
                            offset: const Offset(0, -2),
                          ),
                        ],
                      ),
                      child: const Icon(
                        Icons.menu_book,
                        color: Colors.white,
                        size: 28,
                      ),
                    ),
                  );
                },
              ),
            ),
          ),

          // Inverted scoop cutout mask for center tab
          Positioned(
            bottom: 12,
            child: CustomPaint(
              size: const Size(80, 64),
              painter: _ScoopCutoutPainter(
                backgroundColor: const Color(0xFF0A0A0A),
              ),
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildTab(int index) {
    final isSelected = widget.currentIndex == index;
    final tab = _tabs[index];

    return GestureDetector(
      onTap: () => widget.onTap(index),
      behavior: HitTestBehavior.opaque,
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 200),
        curve: Curves.easeOut,
        padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
        decoration: isSelected
            ? BoxDecoration(
                shape: BoxShape.circle,
                color: const Color(0xFFFF6A00).withOpacity(0.1),
              )
            : null,
        child: AnimatedScale(
          scale: isSelected ? 1.1 : 1.0,
          duration: const Duration(milliseconds: 200),
          curve: Curves.easeOutBack,
          child: Icon(
            tab.icon,
            size: 24,
            color: isSelected
                ? const Color(0xFFFF6A00)
                : const Color(0xFF222222).withOpacity(0.6),
          ),
        ),
      ),
    );
  }
}

class _NavTab {
  final IconData icon;
  final String label;
  final bool isCenter;

  const _NavTab({

// Custom painter for the inverted scoop cutout effect
class _ScoopCutoutPainter extends CustomPainter {
  final Color backgroundColor;

  _ScoopCutoutPainter({required this.backgroundColor});

  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = backgroundColor
      ..style = PaintingStyle.fill;

    final path = Path();

    // Draw the scoop shape - inverted arc in the center
    path.moveTo(0, size.height);
    path.lineTo(0, 0);
    
    // Left side flat
    path.lineTo(size.width * 0.25, 0);
    
    // Scoop curve (inverted arc)
    path.arcToPoint(
      Offset(size.width * 0.75, 0),
      radius: const Radius.circular(28),
      clockwise: false,
    );
    
    // Right side flat
    path.lineTo(size.width, 0);
    path.lineTo(size.width, size.height);
    
    // Bottom flat
    path.lineTo(0, size.height);
    path.close();

    // Cut out the center to reveal the FAB behind
    final scoopPath = Path();
    scoopPath.moveTo(size.width * 0.15, size.height);
    scoopPath.lineTo(size.width * 0.15, size.height * 0.3);
    
    // Smooth scoop arc
    scoopPath.quadraticBezierTo(
      size.width * 0.5, -size.height * 0.5,
      size.width * 0.85, size.height * 0.3,
    );
    
    scoopPath.lineTo(size.width * 0.85, size.height);
    scoopPath.close();

    // Combine: fill the bar, then cut out the scoop
    canvas.drawPath(
      Path.combine(
        PathOperation.difference,
        path,
        scoopPath,
      ),
      paint,
    );
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';
import '../components/tether_bottom_nav.dart';

class MainShell extends StatefulWidget {
  final Widget child;
  final String location;

  const MainShell({
    super.key,
    required this.child,
    required this.location,
  });

  @override
  State<MainShell> createState() => _MainShellState();
}

class _MainShellState extends State<MainShell> {
  int _getCurrentIndex(String location) {
    if (location.startsWith('/clubs') || location == '/clubs') return 0;
    if (location.startsWith('/discover')) return 1;
    if (location.startsWith('/reader') || location.startsWith('/books')) return 2;
    if (location.startsWith('/chat') || location.startsWith('/dms')) return 3;
    if (location.startsWith('/profile')) return 4;
    return 0;
  }

  void _onTabTapped(int index) {
    switch (index) {
      case 0:
        context.go('/clubs');
        break;
      case 1:
        context.go('/discover');
        break;
      case 2:
        context.go('/books');
        break;
      case 3:
        context.go('/dms');
        break;
      case 4:
        context.go('/profile');
        break;
    }
  }

  @override
  Widget build(BuildContext context) {
    final currentIndex = _getCurrentIndex(widget.location);

    return Scaffold(
      backgroundColor: const Color(0xFF0A0A0A),
      body: widget.child,
      bottomNavigationBar: TetherBottomNav(
        currentIndex: currentIndex,
        onTap: _onTabTapped,
      ),
      extendBody: true, // Allows content to extend behind the nav bar
    );
  }
}

class _BookNestBottomNavState extends State<BookNestBottomNav>
    with SingleTickerProviderStateMixin {
  late AnimationController _fabController;
  late Animation<double> _fabScale;

  @override
  void initState() {
    super.initState();
    _fabController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 220),
    );
    _fabScale = Tween<double>(begin: 1.0, end: 1.12).animate(
      CurvedAnimation(parent: _fabController, curve: Curves.easeOutBack),
    );

  @override
  void dispose() {
    _fabController.dispose();
    super.dispose();
  }

  void _onCenterTap() {
    HapticFeedback.mediumImpact();
    _fabController.forward().then((_) => _fabController.reverse());
    widget.onTap(2);
  }

  @override
  Widget build(BuildContext context) {
    return SizedBox(
      height: 100,
      child: Stack(
        alignment: Alignment.bottomCenter,
        clipBehavior: Clip.none,

            height: 64,
            decoration: BoxDecoration(
              color: const Color(0xFF0A0A0A),
              borderRadius: BorderRadius.circular(24),
              boxShadow: [
                BoxShadow(
                  color: Colors.black.withOpacity(0.4),
                  blurRadius: 20,
                  offset: const Offset(0, 8),
                ),
              ],
            ),
            child: Row(
              mainAxisAlignment: MainAxisAlignment.spaceEvenly,
              children: [
                _buildTab(0, Icons.view_agenda_outlined),
                _buildTab(1, Icons.explore_outlined),
                const SizedBox(width: 56), // Center gap
                _buildTab(3, Icons.chat_bubble_outline),
                _buildTab(4, Icons.person_outline),
              ],
            ),
          ),

          // Center FAB
          Positioned(
            top: -8,
            child: GestureDetector(
              onTap: _onCenterTap,
              child: AnimatedBuilder(
                animation: _fabScale,
                builder: (context, child) => Transform.scale(
                  scale: widget.currentIndex == 2 ? _fabScale.value : 1.0,
                  child: Container(
                    width: 56,
                    height: 56,
                    decoration: const BoxDecoration(
                      shape: BoxShape.circle,
                      gradient: LinearGradient(
                        begin: Alignment.topCenter,
                        end: Alignment.bottomCenter,
                        colors: [Color(0xFFFFD000), Color(0xFFFF6A00)],
                      ),
                      boxShadow: [
                        BoxShadow(
                          color: Color(0xFFFF6A00),
                          blurRadius: 16,
                          offset: Offset(0, 6),
                        ),
                      ],
                    ),
                    child: const Icon(
                      Icons.menu_book,
                      color: Colors.white,
                      size: 28,
                    ),
                  ),
                ),
              ),
            ),
          ),

          // Scoop cutout
          Positioned(
            bottom: 12,
            child: CustomPaint(
              size: const Size(80, 64),
              painter: _ScoopPainter(),
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildTab(int index, IconData icon) {
    final isSelected = widget.currentIndex == index;
    return GestureDetector(
      onTap: () => widget.onTap(index),
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 200),
        padding: const EdgeInsets.all(12),
        decoration: isSelected
            ? BoxDecoration(
                shape: BoxShape.circle,
                color: const Color(0xFFFF6A00).withOpacity(0.1),
              )
            : null,
        child: Icon(
          icon,
          size: 24,
          color: isSelected
              ? const Color(0xFFFF6A00)
              : const Color(0xFF222222).withOpacity(0.6),
        ),
      ),
    );
  }
}

class _ScoopPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = const Color(0xFF0A0A0A)
      ..style = PaintingStyle.fill;

    final path = Path()
      ..moveTo(0, size.height)
      ..lineTo(0, 0)
      ..lineTo(size.width * 0.15, 0)
      ..quadraticBezierTo(
        size.width * 0.5, -size.height * 0.5,
        size.width * 0.85, 0,
      )
      ..lineTo(size.width, 0)
      ..lineTo(size.width, size.height)
      ..close();

    canvas.drawPath(path, paint);
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class MainShell extends StatefulWidget {
  final Widget child;
  final String location;

  const MainShell({super.key, required this.child, required this.location});

  @override
  State<MainShell> createState() => _MainShellState();
}

class _MainShellState extends State<MainShell> {
  int _getIndex(String loc) {
    if (loc == '/clubs') return 0;
    if (loc == '/discover') return 1;
    if (loc.startsWith('/books') || loc.startsWith('/reader')) return 2;
    if (loc.startsWith('/chat') || loc.startsWith('/dms')) return 3;
    if (loc == '/profile') return 4;
    return 0;
  }

  void _onTap(int index) {
    switch (index) {
      case 0: context.go('/clubs'); break;
      case 1: context.go('/discover'); break;
      case 2: context.go('/books'); break;
      case 3: context.go('/dms'); break;
      case 4: context.go('/profile'); break;
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFF0A0A0A),
      body: widget.child,
      bottomNavigationBar: BookNestBottomNav(
        currentIndex: _getIndex(widget.location),
        onTap: _onTap,
      ),
      extendBody: true,
    );
  }
}

  final _usernameController = TextEditingController();
  final _phoneController = TextEditingController();
  bool _isLoading = false;

  Future<void> _register() async {
    setState(() {
      _isLoading = true;
      _error = null;
    });

    try {
      final supabase = SupabaseService();
      
      // 1. Sign up with auth
      final response = await supabase.auth.signUp(

          userId: response.user!.id,
          username: _usernameController.text.trim(),
          displayName: _usernameController.text.trim(),
          phoneNumber: _phoneController.text.trim(),

        'content_format': 'html',
        'moderation_status': 'pending',
        'added_by': user.id,
      }).select();
      
      final bookId = bookResponse[0]['id'];
      
      // 2. Create chapter with full content
      await supabase.fromBookChapters.insert({
        'club_book_id': bookId,
        'chapter_number': 1,
        'title': 'Chapter 1',
